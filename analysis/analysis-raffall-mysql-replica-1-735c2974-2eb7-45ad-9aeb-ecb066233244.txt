            t.raffall_id as raffall_id,
            r.raffall_title as raffall_title,
            r.raffall_end_date as raffall_end_date,
            (SELECT COUNT(DISTINCT user_id) FROM raffall_tickets rt WHERE rt.raffall_id = t.raffall_id)
                as num_entrants,
            t.user_id as user_id,
            u.user_name as user_name,
            u.user_email as user_email,
            (select count(DISTINCT atp.user_id)

            as affiliate_entrants,

select count(*) from (
            select U.*,
                   coalesce(O.total_spent, 0) as total_spent,
                   coalesce(O.total_invite, 0) as total_invite,
                   coalesce(O.total_featured, 0) as total_featured,
                   coalesce(O.total_translations, 0) as total_translations,
                   coalesce(T.ticket_revenue, 0) as ticket_revenue,
                   coalesce(RP.raffles_published, 0) as raffles_published,
           (coalesce(O.total_spent, 0) + coalesce(O.total_invite, 0) + coalesce(O.total_featured, 0) + coalesce(O.total_translations, 0)  + coalesce(T.ticket_revenue, 0) + coalesce(U.account_price, 0)) as lifetimevalue,
           (coalesce(O.total_invite, 0) + coalesce(O.total_featured, 0) + coalesce(O.total_translations, 0) + coalesce(U.account_price, 0) ) as high_income_value
            from
                (select u.user_id, u.user_created, um.utm_params, u.account_expiration_date,
                        acs.price as account_price
                        ,
                    hc.host_category_name,
                    CASE
                        WHEN um.utm_params IS NULL THEN "Organic"
                        WHEN um.utm_params IS NOT NULL
                            AND JSON_UNQUOTE(JSON_EXTRACT(um.utm_params, "$.utm_medium")) IN ('paid_social', 'social', 'cpc', 'paid_search', 'ppc', 'ad','ads', 'video_ad', 'paid_video', 'ig_ad') THEN "Advertise"
                        WHEN um.utm_params IS NOT NULL
                            AND (JSON_UNQUOTE(JSON_EXTRACT(um.utm_params, "$.utm_medium")) IN ('email', 'Email') or JSON_UNQUOTE(JSON_EXTRACT(um.utm_params, "$.utm_source")) IN ('email', 'Email')) THEN "Email"
                        ELSE "Organic"
                    END AS source_type
                from users u
                        left join users_marketing um on u.user_id = um.user_id
                        left join account_settings acs on u.account_settings_id = acs.account_settings_id
                        left join host_categories hc on u.host_category_id = hc.host_category_id
                where u.is_hoster = 1
                
                 and DATE(u.user_created) = CURDATE() ) U
                left join (
                    select o.user_id,
                      SUM(CASE WHEN o.product = "TICKET" THEN o.total_price ELSE 0 END) AS total_spent,
                      SUM(CASE WHEN o.product = "INVITE" THEN o.total_price ELSE 0 END) AS total_invite,
                      SUM(CASE WHEN o.product = "FEATURED" THEN o.total_price ELSE 0 END) AS total_featured,
                      SUM(CASE WHEN o.product = "TRANSLATION" THEN o.total_price ELSE 0 END) AS total_translations
                     from `order` o
                     where o.order_status in ("PAID", "PREPAID")

                     and DATE(o.created_date) = CURDATE()  
                    group by o.user_id
                ) O on O.user_id = U.user_id
                left join (
                      select r.user_id, sum(o.total_price) as ticket_revenue from raffalls r
                        left join `order` o on o.raffall_id = r.raffall_id
                      where o.order_status in ("PAID", "PREPAID")
                      and o.product = "TICKET"
                         and DATE(o.created_date) = CURDATE() 
                      group by r.user_id
                ) T on T.user_id = U.user_id
                left join (
                    select r.user_id, count(r.raffall_id) as raffles_published from raffalls r
                    where r.raffall_status in ("Live", "Ended", "Drawn")
                     and r.raffall_created = CURDATE() 
                    group by r.user_id
                ) RP on RP.user_id = U.user_id
