SELECT 
            t.affiliate_commission_request_id as affiliate_commission_request_id,
            t.raffall_id as raffall_id,
            r.raffall_title as raffall_title,
            r.raffall_end_date as raffall_end_date,
            (SELECT COUNT(DISTINCT user_id) FROM raffall_tickets rt WHERE rt.raffall_id = t.raffall_id)
                as num_entrants,
            t.user_id as user_id,
            u.user_name as user_name,
            u.user_email as user_email,
            (select count(DISTINCT atp.user_id)
            from affiliate_ticket_purchased atp
            where atp.affiliate_commission_request_id = t.affiliate_commission_request_id)
            as affiliate_entrants,
            (select SUM(atpa.ticket_count)
            from affiliate_ticket_purchased atpa
            where atpa.affiliate_commission_request_id = t.affiliate_commission_request_id)
            as affiliate_tickets_sold,
            (select (SUM(atpb.commission_rate * atpb.ticket_count) / SUM(atpb.ticket_count)) * 100
            from affiliate_ticket_purchased atpb
            where atpb.affiliate_commission_request_id = t.affiliate_commission_request_id)
            as avg_affiliate_commission,
            t.request_date as withdrawal_requested,
            t.amount as amount,
            t.request_status as request_status,
            cur.currency_sign,
            wcur.currency_name as account_currency_name,
            wcur.currency_sign as account_currency_sign,
            wbd.account_currency_id,
            wbd.type as bank_type,
            wbd.account as account,
            wbd.bank_account as bank_account,
            wbd.sort_code as sort_code,
            wbd.email as email,
            wbd.pay_date as pay_date,
            wbd.account_type as account_type,
            u2.user_name as host_user_pay_name
         FROM `affiliate_commission_request` `t` 
            LEFT JOIN raffalls r on t.raffall_id = r.raffall_id
            LEFT JOIN users u on t.user_id = u.user_id
            INNER JOIN withdrawal_bank_details wbd on wbd.withdrawal_bank_details_id = t.withdrawal_bank_details_id
            LEFT JOIN currencies cur on cur.currency_id = r.raffall_currency
            LEFT JOIN users u2 on wbd.host_pay_user_id = u2.user_id
            LEFT JOIN currencies wcur on wcur.currency_id = wbd.account_currency_id
         WHERE t.affiliate_commission_request_id IS NOT NULL GROUP BY t.affiliate_commission_request_id, r.raffall_id ORDER BY t.affiliate_commission_request_id asc\G
